-- SUPABASE-READY LOGISTICS MARKETPLACE SCHEMA (Postgres + PostGIS)
-- - Uses Supabase Auth (auth.users) + public.profiles
-- - Map-based pickup/delivery via geography(Point,4326)
-- - Lat/Lng <-> Point sync triggers
-- - updated_at triggers
-- - Offer + Shipment status history triggers
-- - Offer status enforcement (customer accepts/rejects, company manages)
-- - RLS policies using auth.uid()
-- =========================================================

begin;

begin
  if not exists (select 1 from pg_type where typname = 'announcement_status') then
    create type public.announcement_status as enum ('pending','assigned','in_transit','delivered','cancelled');
  end if;

  if not exists (select 1 from pg_type where typname = 'offer_status') then
    create type public.offer_status as enum ('pending','accepted','rejected','cancelled','expired');
  end if;

  if not exists (select 1 from pg_type where typname = 'shipment_status') then
    create type public.shipment_status as enum ('assigned','picked_up','in_transit','delivered','cancelled');
  end if;

  if not exists (select 1 from pg_type where typname = 'document_status') then
    create type public.document_status as enum ('pending','approved','rejected','expired');
  end if;

  if not exists (select 1 from pg_type where typname = 'payment_status') then
    create type public.payment_status as enum ('pending','paid','failed','refunded','cancelled');
  end if;

  if not exists (select 1 from pg_type where typname = 'payout_status') then
    create type public.payout_status as enum ('pending','processing','paid','failed','cancelled');
  end if;

  if not exists (select 1 from pg_type where typname = 'device_type') then
    create type public.device_type as enum ('ios','android','web','unknown');
  end if;

  if not exists (select 1 from pg_type where typname = 'company_user_role') then
    create type public.company_user_role as enum ('owner','admin','dispatcher','driver_manager','accountant','viewer');
  end if;
end $$;

create or replace function public.current_customer_id()
returns bigint
language sql
stable
set search_path = public
as $$
  select c.id
  from public.customers c
  where c.user_id = auth.uid()
  limit 1;
$$;

create or replace function public.is_company_member(p_company_id bigint)
returns boolean
language sql
stable
set search_path = public
as $$
  select exists (
    select 1
    from public.company_users cu
    where cu.company_id = p_company_id
      and cu.user_id = auth.uid()
  );
$$;

create or replace function public.is_company_admin(p_company_id bigint)
returns boolean
language sql
stable
set search_path = public
as $$
  select exists (
    select 1
    from public.company_users cu
    where cu.company_id = p_company_id
      and cu.user_id = auth.uid()
      and cu.role in ('owner','admin')
  );
$$;

-- updated_at standard trigger
create or replace function public._set_updated_at()
returns trigger
language plpgsql
set search_path = public
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

do $$
declare
  t text;
begin
  foreach t in array array[
    'profiles','companies','company_users','customers','drivers','vehicles',
    'announcements','offers','shipments','payments','documents','driver_documents',
    'driver_sessions','system_settings','wallets'
  ]
  loop
    execute format('drop trigger if exists trg_%I_updated_at on public.%I;', t, t);
    execute format('create trigger trg_%I_updated_at before update on public.%I for each row execute function public._set_updated_at();', t, t);
  end loop;
end $$;

-- Sync: announcements Point <-> lat/lng
create or replace function public._sync_announcement_points()
returns trigger
language plpgsql
set search_path = public
as $$
begin
  if (new.pickup_point is null) and (new.pickup_lat is not null) and (new.pickup_lng is not null) then
    new.pickup_point := ST_SetSRID(ST_MakePoint(new.pickup_lng, new.pickup_lat), 4326)::geography;
  end if;

  if new.pickup_point is not null then
    new.pickup_lat := ST_Y(new.pickup_point::geometry);
    new.pickup_lng := ST_X(new.pickup_point::geometry);
  end if;

  if (new.delivery_point is null) and (new.delivery_lat is not null) and (new.delivery_lng is not null) then
    new.delivery_point := ST_SetSRID(ST_MakePoint(new.delivery_lng, new.delivery_lat), 4326)::geography;
  end if;

  if new.delivery_point is not null then
    new.delivery_lat := ST_Y(new.delivery_point::geometry);
    new.delivery_lng := ST_X(new.delivery_point::geometry);
  end if;

  if new.pickup_point is null then
    raise exception 'pickup_point (or pickup_lat/pickup_lng) is required';
  end if;

  if new.delivery_point is null then
    raise exception 'delivery_point (or delivery_lat/delivery_lng) is required';
  end if;

  return new;
end;
$$;

drop trigger if exists trg_sync_announcement_points on public.announcements;
create trigger trg_sync_announcement_points
before insert or update of
  pickup_point, pickup_lat, pickup_lng,
  delivery_point, delivery_lat, delivery_lng
on public.announcements
for each row
execute function public._sync_announcement_points();

-- Sync: shipment_tracking point <-> lat/lng
create or replace function public._sync_shipment_tracking_point()
returns trigger
language plpgsql
set search_path = public
as $$
begin
  if (new.point is null) then
    new.point := ST_SetSRID(ST_MakePoint(new.lng, new.lat), 4326)::geography;
  end if;

  new.lat := ST_Y(new.point::geometry);
  new.lng := ST_X(new.point::geometry);

  return new;
end;
$$;

drop trigger if exists trg_sync_shipment_tracking_point on public.shipment_tracking;
create trigger trg_sync_shipment_tracking_point
before insert or update of lat, lng, point
on public.shipment_tracking
for each row
execute function public._sync_shipment_tracking_point();

-- Offer status enforcement:
-- - Company members can set any status
-- - Customer (owner of announcement) can only: pending -> accepted/rejected (and responded_at auto)
create or replace function public._enforce_offer_status_update()
returns trigger
language plpgsql
set search_path = public
as $$
declare
  is_company boolean;
  is_customer_owner boolean;
begin
  if new.status is distinct from old.status then
    is_company := public.is_company_member(old.company_id);

    if is_company then
      -- company-side: ok
      return new;
    end if;

    is_customer_owner := exists (
      select 1
      from public.announcements a
      where a.id = old.announcement_id
        and a.customer_id = public.current_customer_id()
    );

    if not is_customer_owner then
      raise exception 'not allowed to update offer status';
    end if;

    if old.status <> 'pending' then
      raise exception 'offer status can only be responded once (must be pending)';
    end if;

    if new.status not in ('accepted','rejected') then
      raise exception 'customer can only set offer status to accepted or rejected';
    end if;

    new.responded_at := now();
  end if;

  return new;
end;
$$;

drop trigger if exists trg_enforce_offer_status_update on public.offers;
create trigger trg_enforce_offer_status_update
before update of status
on public.offers
for each row
execute function public._enforce_offer_status_update();

-- Create shipment when offer accepted + mark announcement assigned
create or replace function public._create_shipment_on_offer_accepted()
returns trigger
language plpgsql
set search_path = public
as $$
declare
  a public.announcements%rowtype;
begin
  if new.status = 'accepted' and (old.status is distinct from new.status) then
    if exists (select 1 from public.shipments s where s.offer_id = new.id) then
      return new;
    end if;

    select * into a from public.announcements where id = new.announcement_id;

    insert into public.shipments (
      offer_id, announcement_id, customer_id, company_id, driver_id, vehicle_id,
      tracking_code, status, final_price, currency, created_at, updated_at
    )
    values (
      new.id, new.announcement_id, a.customer_id, new.company_id, new.driver_id, null,
      encode(gen_random_bytes(6), 'hex'),
      'assigned',
      new.price, new.currency,
      now(), now()
    );

    update public.announcements
      set status = 'assigned',
          company_id = new.company_id
    where id = new.announcement_id;
  end if;

  return new;
end;
$$;

drop trigger if exists trg_create_shipment_on_offer_accepted on public.offers;
create trigger trg_create_shipment_on_offer_accepted
after update of status
on public.offers
for each row
execute function public._create_shipment_on_offer_accepted();

-- Auto-create profile + customer on signup (Supabase standard)
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.profiles (id, email, created_at, updated_at)
  values (new.id, new.email, now(), now())
  on conflict (id) do nothing;

  insert into public.customers (user_id, created_at, updated_at)
  values (new.id, now(), now())
  on conflict (user_id) do nothing;

  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
after insert on auth.users
for each row execute function public.handle_new_user();
